cmake_minimum_required(VERSION 3.20)
project(rudy-dns-trace VERSION 0.1.0 LANGUAGES C)

option(RDT_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

add_library(rudy_dns_trace_core src/rudy_dns_trace.c)
target_include_directories(rudy_dns_trace_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_features(rudy_dns_trace_core PUBLIC c_std_11)

add_executable(rudy-dns-trace src/main.c)
target_link_libraries(rudy-dns-trace PRIVATE rudy_dns_trace_core)
target_compile_features(rudy-dns-trace PRIVATE c_std_11)

include(CTest)
if(BUILD_TESTING)
    add_executable(rudy-dns-trace-tests tests/test.c)
    target_link_libraries(rudy-dns-trace-tests PRIVATE rudy_dns_trace_core)
    target_compile_features(rudy-dns-trace-tests PRIVATE c_std_11)
    add_test(NAME parser-regressions COMMAND rudy-dns-trace-tests)
    set_tests_properties(parser-regressions PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
    add_test(NAME cli-help COMMAND rudy-dns-trace --help)
endif()

if(MSVC)
    target_compile_options(rudy_dns_trace_core PRIVATE /W4)
    target_compile_options(rudy-dns-trace PRIVATE /W4)
    if(BUILD_TESTING)
        target_compile_options(rudy-dns-trace-tests PRIVATE /W4)
    endif()
else()
    target_compile_options(rudy_dns_trace_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    target_compile_options(rudy-dns-trace PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    if(BUILD_TESTING)
        target_compile_options(rudy-dns-trace-tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
    if(RDT_ENABLE_SANITIZERS)
        target_compile_options(rudy_dns_trace_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_compile_options(rudy-dns-trace PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(rudy-dns-trace PRIVATE -fsanitize=address,undefined)
        if(BUILD_TESTING)
            target_compile_options(rudy-dns-trace-tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(rudy-dns-trace-tests PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endif()

install(TARGETS rudy-dns-trace RUNTIME DESTINATION bin)
install(FILES README.md LICENSE SECURITY.md CONTRIBUTING.md DESTINATION share/doc/rudy-dns-trace)
install(DIRECTORY docs/ DESTINATION share/doc/rudy-dns-trace/docs FILES_MATCHING PATTERN "*.md")
